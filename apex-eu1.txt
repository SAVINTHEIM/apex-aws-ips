
name: Update Apex AWS IP Candidates

on:
  schedule:
    - cron: '0 4 * * *'
  workflow_dispatch:

permissions:
  contents: write

concurrency:
  group: update-apex-aws-ips
  cancel-in-progress: false

jobs:
  update:
    runs-on: ubuntu-latest

    steps:
      - name: Checkout repo
        uses: actions/checkout@v4

      - name: Fetch and filter AWS IPv4 ranges
        shell: bash
        run: |
          set -euo pipefail

          URL="https://ip-ranges.amazonaws.com/ip-ranges.json"
          TMP_JSON="$(mktemp)"
          TMP_IPS="$(mktemp)"
          trap 'rm -f "$TMP_JSON" "$TMP_IPS"' EXIT

          curl --fail --silent --show-error --location \
            --retry 3 --retry-delay 2 \
            "$URL" -o "$TMP_JSON"

          jq -e '
            (.prefixes | type == "array") and
            (.syncToken | type == "string") and
            (.createDate | type == "string")
          ' "$TMP_JSON" > /dev/null

          jq -r '
            .prefixes[]
            | select(
                .service == "EC2" and
                (
                  .region == "eu-central-1" or
                  .region == "eu-west-1" or
                  .region == "eu-west-2" or
                  .region == "eu-west-3" or
                  .region == "eu-north-1" or
                  .region == "eu-south-1" or
                  .region == "eu-south-2" or
                  .region == "eu-central-2"
                )
              )
            | .ip_prefix
          ' "$TMP_JSON" | sort -u > "$TMP_IPS"

          COUNT="$(wc -l < "$TMP_IPS" | tr -d ' ')"

          if [ "$COUNT" -eq 0 ]; then
            echo "::error::No AWS IPv4 ranges found. Existing list preserved."
            exit 1
          fi

          cp "$TMP_IPS" apex-eu.txt

          echo "AWS publication date: $(jq -r '.createDate' "$TMP_JSON")"
          echo "Unique IPv4 CIDRs: $COUNT"

      - name: Commit and push if changed
        uses: stefanzweifel/git-auto-commit-action@v5
        with:
          commit_message: "Update European AWS IPv4 candidates"
          file_pattern: apex-eu.txt
